cmake_minimum_required(VERSION 3.20)
project(LibertyGenesis VERSION 0.1.0 LANGUAGES CXX)
add_library(genesis_engine INTERFACE)
target_include_directories(genesis_engine INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(genesis_engine INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(genesis_smoke tests/smoke.cpp)
    target_link_libraries(genesis_smoke PRIVATE genesis_engine)
    add_test(NAME genesis_smoke COMMAND genesis_smoke)
endif()
