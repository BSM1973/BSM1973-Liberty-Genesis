cmake_minimum_required(VERSION 3.20)
project(LibertyGenesis VERSION 0.1.0 LANGUAGES CXX)
add_library(genesis_engine INTERFACE)
target_include_directories(genesis_engine INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(genesis_engine INTERFACE cxx_std_17)
add_executable(genesis_demo tools/render_demo.cpp)
target_link_libraries(genesis_demo PRIVATE genesis_engine)
add_executable(genesis_presets tools/render_presets.cpp)
target_link_libraries(genesis_presets PRIVATE genesis_engine)
include(CTest)
if(BUILD_TESTING)
    add_executable(genesis_smoke tests/smoke.cpp)
    target_link_libraries(genesis_smoke PRIVATE genesis_engine)
    add_test(NAME genesis_smoke COMMAND genesis_smoke)
    add_executable(genesis_regression tests/regression.cpp)
    target_link_libraries(genesis_regression PRIVATE genesis_engine)
    add_test(NAME genesis_regression COMMAND genesis_regression)
    add_executable(genesis_mpc_midi_adapter tests/mpc_midi_adapter.cpp)
    target_link_libraries(genesis_mpc_midi_adapter PRIVATE genesis_engine)
    add_test(NAME genesis_mpc_midi_adapter COMMAND genesis_mpc_midi_adapter)
endif()
